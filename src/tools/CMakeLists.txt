CMAKE_MINIMUM_REQUIRED(VERSION 3.10)

PROJECT(${TOOL_PROJECT_NAME} C)

FILE(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.c")

ADD_LIBRARY(${PROJECT_NAME}-lib-dyn     SHARED ${SOURCES})
ADD_LIBRARY(${PROJECT_NAME}-lib-dyn.1   SHARED ${SOURCES})

target_link_libraries(${PROJECT_NAME}-lib-dyn PRIVATE
    $<LINK_LIBRARY:WHOLE_ARCHIVE,
        ${WRAPPER_PROJECT_NAME}-lib-static;
        ${EXT_PROJECT_NAME}-lib-static;
        ${CORE_PROJECT_NAME}-lib-static>
)

TARGET_LINK_LIBRARIES(${PROJECT_NAME}-lib-dyn.1  ${WRAPPER_PROJECT_NAME}-lib-dyn    ${EXT_PROJECT_NAME}-lib-dyn     ${CORE_PROJECT_NAME}-lib-dyn)

SET_TARGET_PROPERTIES(${PROJECT_NAME}-lib-dyn       PROPERTIES OUTPUT_NAME ${PROJECT_NAME})
SET_TARGET_PROPERTIES(${PROJECT_NAME}-lib-dyn.1     PROPERTIES OUTPUT_NAME ${PROJECT_NAME}.1)
